快速验证命令

# Windows 检查证书链与在线 CRL 吊销状态
certutil -verify -urlfetch <certificate.cer>

# 清除本地 CRL 缓存
certutil -urlcache crl delete

# 查看 RFC 3161 可信时间戳 (.tsr) 详情
openssl ts -reply -in <file.tsr> -text

# 校验文件与时间戳签名（需提供对应 TSA 的 CA 证书）
openssl ts -verify -in <file.tsr> -data <file> -CAfile <tsa-ca.pem>
